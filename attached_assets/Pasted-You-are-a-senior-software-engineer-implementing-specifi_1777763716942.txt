7. EXTERNAL API DOMAIN WHITELIST
Create /server/lib/safeFetch.js:

const ALLOWED_DOMAINS = [
  'eutils.ncbi.nlm.nih.gov',
  'api.semanticscholar.org', 
  'api.crossref.org',
  'commons.wikimedia.org',
  'api.openalex.org',
  'api.core.ac.uk',
  'export.arxiv.org',
  'api.unpaywall.org',
  'doaj.org',
  'api.europepmc.org'
];

export async function safeFetch(url, options = {}) {
  const parsed = new URL(url);
  if (!ALLOWED_DOMAINS.includes(parsed.hostname)) {
    throw new Error(`Blocked fetch to unlisted domain: ${parsed.hostname}`);
  }
  // Block private IP ranges
  const ip = parsed.hostname;
  if (/^(localhost|127\.|10\.|192\.168\.|172\.(1[6-9]|2\d|3[01])\.)/.test(ip)) {
    throw new Error(`Blocked fetch to private IP range: ${ip}`);
  }
  return fetch(url, {
    ...options,
    signal: AbortSignal.timeout(10000), // 10 second timeout on all fetches
    headers: { 
      'User-Agent': 'ScholarForge/1.0 (academic research tool; open source)',
      ...options.headers 
    }
  });
}

Replace ALL uses of fetch() in /server/lib/ with safeFetch().

8. BIBTEX AND DOCX INJECTION PROTECTION
In /server/lib/citations.js, add sanitization before any export:

function sanitizeBibtex(str) {
  if (!str) return '';
  return str
    .replace(/[\\{}@]/g, ' ')  // BibTeX special chars
    .replace(/[^\x20-\x7E\u00C0-\u024F]/g, '') // non-printable chars
    .trim();
}

Apply sanitizeBibtex() to every field value before inserting into 
.bib output: title, authors, journal, publisher, url, doi, note.

For .docx export, sanitize all strings through:
function sanitizeDocx(str) {
  if (!str) return '';
  return str
    .replace(/[\x00-\x08\x0B\x0C\x0E-\x1F\x7F]/g, '') // control chars
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .trim();
}
